  function speakWave(w, ms) {
    const bars = $$('b', w); const end = performance.now() + (reduce ? 0 : ms);
    (function tick() { const on = performance.now() < end; bars.forEach(b => b.style.height = on ? (25 + Math.random() * 75) + '%' : '35%'); if (on) setTimeout(tick, 90); })();
  }

  async function runStep(text) {
    if (busy || step >= STEPS.length) return; busy = true;
    const S = STEPS[step], mic = $('#talkMic'), state = $('#micState');
    const me = document.createElement('div'); me.className = 'say say--me';
    convo.appendChild(me);
    if (text == null) {
      mic.classList.add('listening'); state.innerHTML = '<span class="wave" style="color:var(--hv)">' + '<b></b>'.repeat(7) + '</span>';
      speakWave($('.wave', state), 2400);
      const words = S.me.split(' ');
      for (let i = 1; i <= words.length; i++) {
        me.innerHTML = words.slice(0, i).join(' ') + (i < words.length ? ' <span class="live">…</span>' : '');
        convo.scrollTop = convo.scrollHeight; await wait(70 + Math.random() * 60);
      }
